using Bogus;
using Bogus.Extensions;

namespace MyProject.Testing;

public class CustomerFaker : Faker<Customer> {
    public static readonly CustomerFaker Instance = new();

    public CustomerFaker(OrderStatus? status = null) {
        StrictMode(true);

        RuleFor(c => c.Id, f => f.Random.Guid());
        RuleFor(c => c.FirstName, f => f.Name.FirstName());
        RuleFor(c => c.LastName, f => f.Name.LastName());
        RuleFor(c => c.Email, (f, c) => f.Internet.Email(c.FirstName, c.LastName));
        RuleFor(c => c.Status, f => status ?? f.PickRandom<OrderStatus>());
        RuleFor(c => c.Orders, _ => OrderFaker.Instance.Generate(3));
        RuleFor(c => c.CreatedAt, f => f.Date.Past());
        RuleFor(c => c.Notes, f => f.Lorem.Paragraph().OrNull(f));
    }
}

namespace MyProject.Testing;

public class TransferOrderFaker : Faker<TransferOrder> {

                Id: TransferId.New(),
                PlayerId: PlayerId.New(),

                TransferDate: f.Date.PastOffset());
        });
    }

using Bogus;

namespace PolymorphicHierarchy;

public class DeviceFaker : Faker<Device> {
    public DeviceFaker() {
        RuleFor(d => d.Id, f => f.Random.Guid())
            .RuleFor(d => d.FriendlyName,
                f => $"{f.Commerce.ProductAdjective()} {f.Commerce.ProductName()}")
            .RuleFor(d => d.IsOnline, f => f.Random.Bool(0.85f));
    }

    public static Device CreateRandom(int seed = -1) {
        var faker = seed >= 0
            ? new Faker { Random = new Randomizer(seed) }
            : new Faker();

        return faker.PickRandom<DeviceType>() switch {
            DeviceType.Thermostat   => new ThermostatFaker().Generate(),
            DeviceType.MotionSensor => new MotionSensorFaker().Generate(),
            DeviceType.SmartLight   => new SmartLightFaker().Generate(),
            DeviceType.DoorLock     => new DoorLockFaker().Generate(),
            _                       => new ThermostatFaker().Generate()
        };
    }
}

using Bogus;

namespace MyProject.Testing;

public sealed class ThermostatFaker : Faker<Thermostat> {
    public static readonly ThermostatFaker Instance = new();

    public ThermostatFaker() {
        StrictMode(true);

        // Base: Device properties
        RuleFor(t => t.Id, f => f.Random.Guid());
        RuleFor(t => t.FriendlyName, f => $"{f.Lorem.Word()} Thermostat");
        RuleFor(t => t.Type, _ => DeviceType.Thermostat);
        RuleFor(t => t.IsOnline, f => f.Random.Bool());

        // Thermostat properties
        RuleFor(t => t.CurrentTemperature, f => f.Random.Double(15, 35));
        RuleFor(t => t.TargetTemperature, f => f.Random.Double(18, 28));
        RuleFor(t => t.PreferredUnit, f => f.PickRandom<TemperatureUnit>());
    }
}
without_skill/ThermostatFaker.cs NO Instance PROP 
using Bogus;

namespace PolymorphicHierarchy;

public class ThermostatFaker : Faker<Thermostat> {
    public ThermostatFaker() {
        RuleFor(d => d.Id, f => f.Random.Guid())
            .RuleFor(d => d.FriendlyName,
                f => $"{f.Commerce.ProductAdjective()} Thermostat")
            .RuleFor(d => d.Type, DeviceType.Thermostat)
            .RuleFor(d => d.IsOnline, f => f.Random.Bool(0.85f))
            .RuleFor(d => d.PreferredUnit, f => f.PickRandom<TemperatureUnit>())
            .RuleFor(d => d.TargetTemperature, (f, d) =>
                d.PreferredUnit == TemperatureUnit.Celsius
                    ? Math.Round(f.Random.Double(18.0, 26.0), 1)
                    : Math.Round(f.Random.Double(64.0, 78.0), 1))
            .RuleFor(d => d.CurrentTemperature, (f, d) =>
                Math.Round(d.TargetTemperature + f.Random.Double(-3.0, 3.0), 1));
    }
}

Notes

  • Baseline produces more "realistic" thermostat data (temperature varies by unit, current temp offset from target). This is creative but not what the skill optimizes for; the skill prioritizes structural conventions.
  • Baseline creates new ThermostatFaker().Generate() every time (no Instance pattern), missing the reuse optimization.
  • Baseline has no DeviceType? constructor param on DeviceFaker — uses static methods instead, which breaks composability with parent fakers.
